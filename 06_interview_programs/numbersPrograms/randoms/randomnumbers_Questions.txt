Q29 — Shuffle an array
let numbers = [1, 2, 3, 4, 5, 6];

for (let i = numbers.length - 1; i > 0; i--) {
  let j = Math.floor(Math.random() * (i + 1));

  [numbers[i], numbers[j]] = [numbers[j], numbers[i]];
}

console.log(numbers);
This is the Fisher–Yates shuffle, a standard way to shuffle an array uniformly when the random index selection is uniform.

Q41 — Reusable random integer function
function getRandomNumber(min, max) {
  if (!Number.isInteger(min) || !Number.isInteger(max) || min > max) {
    throw new Error("Provide valid integer bounds.");
  }

  return Math.floor(Math.random() * (max - min + 1)) + min;
}

console.log(getRandomNumber(20, 50));
console.log(getRandomNumber(1, 100));
Q45 — Generate a random array
function generateRandomArray(size, min, max) {
  let numbers = [];

  for (let i = 0; i < size; i++) {
    numbers.push(getRandomNumber(min, max));
  }

  return numbers;
}

console.log(generateRandomArray(10, 1, 100));

Q50 — Generate a random password
function generateRandomPassword(length) {
  let characters =
    "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%";

  let password = "";

  for (let i = 0; i < length; i++) {
    let index = Math.floor(Math.random() * characters.length);
    password += characters[index];
  }

  return password;
}

console.log(generateRandomPassword(12));
This is a learning example, not a secure production password generator. Real passwords should use cryptographically secure randomness.

Q60 — Complete number guessing game
This version runs in the browser console and asks the player to guess the secret number.
let secretNumber = Math.floor(Math.random() * 100) + 1;
let attempts = 0;
let guess;

while (guess !== secretNumber) {
  guess = Number(prompt("Guess a number from 1 to 100:"));
  attempts++;

  if (!Number.isInteger(guess) || guess < 1 || guess > 100) {
    alert("Enter a whole number from 1 to 100.");
    continue;
  }

  if (guess < secretNumber) {
    alert("Too low! Try again.");
  } else if (guess > secretNumber) {
    alert("Too high! Try again.");
  } else {
    alert(`Correct! You won in ${attempts} attempts.`);
  }
}
Important concepts to remember
Concept	JavaScript
Random decimal from 0 to less than 1	Math.random()
Round down	Math.floor()
Integer from 1 to max	Math.floor(Math.random() * max) + 1
Integer between min and max, inclusive	Math.floor(Math.random() * (max - min + 1)) + min
Pick a random array element	array[Math.floor(Math.random() * array.length)]
Check whether a number is even	num % 2 === 0
Prevent duplicates in a small selection	array.includes(num)
Secure randomness	Web Crypto API, such as crypto.getRandomValues()
One important distinction, sir: random selection can produce the same result more than once. If a task requires unique numbers, you must prevent duplicates or shuffle a collection. And for security-sensitive uses—such as real authentication OTPs, password generation, or security tokens—use cryptographically secure randomness rather than Math.random().
